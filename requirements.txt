import streamlit as st

st.set_page_config(page_title="IBM Bob Copilot", layout="wide")

st.title("🧠 IBM Bob Intelligent Copilot")

if "chat" not in st.session_state:
    st.session_state.chat = []

mode = st.sidebar.selectbox("Mode", ["Repo Brain", "Incident Debug", "Workflow Builder"])

user_input = st.chat_input("Ask IBM Bob...")

def render_chat():
    for msg in st.session_state.chat:
        with st.chat_message(msg["role"]):
            st.write(msg["content"])

render_chat()

if user_input:
    st.session_state.chat.append({"role": "user", "content": user_input})

    response = f"🤖 Bob ({mode}) processed: {user_input}"

    st.session_state.chat.append({"role": "assistant", "content": response})

    st.rerun()
